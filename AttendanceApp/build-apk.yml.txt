name: Build Android APK

on:
  workflow_dispatch:      # "Run workflow" button
  push:
    branches: [main]

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '17'

      - name: Face model check
        env:
          MODEL_URL: ${{ vars.MODEL_URL }}
        run: |
          F=app/src/main/assets/mobilefacenet.tflite
          if [ ! -f "$F" ] && [ -n "$MODEL_URL" ]; then
            curl -L --fail -o "$F" "$MODEL_URL"
          fi
          if [ ! -f "$F" ]; then
            echo "::error::mobilefacenet.tflite is missing. Upload it to app/src/main/assets/ (or set a repository variable MODEL_URL)."
            exit 1
          fi
          ls -l "$F"

      - name: Accept Android SDK licenses
        run: yes | "$ANDROID_HOME/cmdline-tools/latest/bin/sdkmanager" --licenses > /dev/null || true

      - uses: gradle/actions/setup-gradle@v4
        with:
          gradle-version: '8.7'

      - name: Build release APK
        run: gradle assembleRelease --no-daemon --stacktrace

      - uses: actions/upload-artifact@v4
        with:
          name: stanford-attendance-apk
          path: app/build/outputs/apk/release/*.apk
